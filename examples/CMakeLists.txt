cmake_minimum_required(VERSION 3.24)
project(ISpMVExamples LANGUAGES C CXX)
find_package(ISpMV 0.1.0 CONFIG REQUIRED COMPONENTS runtime)
find_package(CUDAToolkit REQUIRED)
add_executable(ispmv_host_c host.c)
target_compile_features(ispmv_host_c PRIVATE c_std_11)
target_link_libraries(ispmv_host_c PRIVATE ISpMV::runtime CUDA::cudart m)
add_executable(ispmv_host_cpp host.cpp)
target_compile_features(ispmv_host_cpp PRIVATE cxx_std_17)
target_link_libraries(ispmv_host_cpp PRIVATE ISpMV::runtime CUDA::cudart)
option(ISPMV_EXAMPLE_DEVICE "Build the CUDA device-buffer example" OFF)
if(ISPMV_EXAMPLE_DEVICE)
    add_executable(ispmv_device device.cpp)
    target_compile_features(ispmv_device PRIVATE cxx_std_17)
    target_link_libraries(ispmv_device PRIVATE ISpMV::runtime CUDA::cudart)
endif()
